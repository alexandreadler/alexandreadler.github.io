Orientação: utilize os exemplos da aula como referência. Nesta atividade, trabalhe com funções simples para as operações da fila. Na fila estática, utilize vetor e MAX = 5. Na fila dinâmica, utilize nós e malloc().
📎 Ver código de referência - Fila Estática 
#include <stdio.h>

#define MAX 5


//=========================================================
// ESTRUTURA DA FILA
//=========================================================

struct Fila {

    // Vetor que armazena os elementos da fila.
    int dados[MAX];

    // Índice do primeiro elemento da fila.
    int inicio;

    // Índice da posição onde será feita a próxima inserção.
    int final;

    // Quantidade de elementos atualmente na fila.
    int quantidade;
};


//=========================================================
// ENQUEUE
// Insere um elemento no final da fila.
//=========================================================

void enqueue(struct Fila *fila, int valor)
{
    // Verifica se a fila está cheia.
    if (fila->quantidade == MAX) {

        printf("Fila cheia!\n");

        return;
    }

    // Insere o valor na posição indicada por final.
    fila->dados[fila->final] = valor;

    //Avança para a próxima posição do vetor.
    //Quando chegar ao final do vetor, faz o índice
    //voltar para 0.
    fila->final = fila->final + 1;
        if (fila->final == MAX) {
            fila->final = 0; // Volta para o início
            }

    // Aumenta a quantidade de elementos.
    fila->quantidade++;
}


//=========================================================
// DEQUEUE
// Remove e retorna o primeiro elemento da fila.
//=========================================================

int dequeue(struct Fila *fila)
{
    // Verifica se a fila está vazia.
    if (fila->quantidade == 0) {

        printf("Fila vazia!\n");

        return -1; //O retorno de -1 em funções como essa é uma convenção padrão na programação (principamente em C) 
        //para indicar que ocorreu um erro ou uma condição excepcional - no caso deste código, 
        //a tentativa de remover um elemento de uma fila vazia.
    }

    // Guarda o valor que será removido.
    int valor = fila->dados[fila->inicio];

    //Avança para a próxima posição do vetor.
    //Quando chegar ao final do vetor, faz o índice voltar para 0.
    fila->inicio = fila->inicio + 1;
        if (fila->inicio == MAX) {
            fila->inicio = 0; // Volta para o início
            }

    // Diminui a quantidade de elementos.
    fila->quantidade--;

    // Retorna o valor removido.
    return valor;
}


//=========================================================
// PEEK
// Consulta o primeiro elemento sem removê-lo.
//=========================================================

int peek(struct Fila *fila)
{
    // Verifica se a fila está vazia.
    if (fila->quantidade == 0) {

        printf("Fila vazia!\n");

        return -1;
    }

    // Retorna o valor do primeiro elemento.
    return fila->dados[fila->inicio];
}


//=========================================================
// SIZE
// Retorna a quantidade de elementos da fila.
//=========================================================

int size(struct Fila *fila)
{
    // Retorna a quantidade armazenada na fila.
    return fila->quantidade;
}


//=========================================================
// ISEMPTY
// Verifica se a fila está vazia.
//=========================================================

int isEmpty(struct Fila *fila)
{
    // Retorna 1 se a quantidade for 0.
    // Caso contrário, retorna 0.
    return fila->quantidade == 0;
}


//=========================================================
// ISFULL
// Verifica se a fila está cheia.
//=========================================================

int isFull(struct Fila *fila)
{
    // Retorna 1 se a quantidade atingir MAX.
    // Caso contrário, retorna 0.
    return fila->quantidade == MAX;
}

//=========================================================
// EXIBIR FILA
// Imprime todos os elementos do início ao fim.
//=========================================================
void exibirFila(struct Fila *fila) {
    //1. Verifica se a fila está vazia
    if (fila->quantidade == 0) {
        printf("Fila vazia!\n");
        return;
    }

    printf("Elementos da fila: ");
    
    //2. Começa no índice onde está o primeiro elemento
    int atual = fila->inicio;

    //3. Repete o ciclo com base na quantidade de elementos
    for (int i = 0; i < fila->quantidade; i++) {
        printf("[%d] ", fila->dados[atual]); // Imprime o elemento atual
        
        //4. Avança para a próxima posição. 
        //Se chegar no fim do vetor (MAX), volta para o zero.
        atual = atual + 1;
        if (atual == MAX) {
            atual = 0;
        }
    }
    
    printf("\n");
}


//=========================================================
// MAIN
//=========================================================

int main()
{
    // Cria uma fila.
    struct Fila fila;

    // A fila começa vazia.
    fila.inicio = 0;
    fila.final = 0;
    fila.quantidade = 0;


    //=====================================================
    // INSERINDO ELEMENTOS
    //=====================================================

    // Insere 10 no final da fila.
    enqueue(&fila, 10);

    // Insere 20 no final da fila.
    enqueue(&fila, 20);

    // Insere 30 no final da fila.
    enqueue(&fila, 30);


    //=====================================================
    // PEAK - CONSULTANDO O PRIMEIRO ELEMENTO
    //=====================================================

    // Consulta o primeiro elemento sem removê-lo.
    printf("Primeiro elemento: %d\n",
           peek(&fila));


    //=====================================================
    // SIZE - CONSULTANDO A QUANTIDADE
    //=====================================================

    // Mostra quantos elementos existem na fila.
    printf("Quantidade de elementos: %d\n",
           size(&fila));

    //=====================================================
    //ISEMPTY - Verifica se a fila está vazia
    //=====================================================

    if (isEmpty(&fila)) {

        printf("A fila esta vazia.\n");

    } else {

        printf("A fila nao esta vazia.\n");
    }


    //=====================================================
    //ISFULL - Verifica se a fila está cheia
    //=====================================================

    if (isFull(&fila)) {

        printf("A fila esta cheia.\n");

    } else {

        printf("A fila nao esta cheia.\n");
    }

    //=====================================================
    // REMOVENDO UM ELEMENTO - DEQUEUE
    //=====================================================

    // Remove o primeiro elemento da fila.
    printf("Removido: %d\n",
           dequeue(&fila));
    
    // Mostra quantos elementos existem na fila agora.
    printf("Quantidade de elementos agora: %d\n",
           size(&fila));
    
    //Exibe os elementos da fila após a remoção
    exibirFila(&fila); 

    return 0;
}

#include <stdio.h>
#include <stdlib.h>


//=========================================================
// ESTRUTURA DO NÓ
//=========================================================

struct No {

    // Valor armazenado no nó.
    int dado;

    // Endereço do próximo nó.
    struct No *proximo;
};


//=========================================================
// ESTRUTURA DA FILA
//=========================================================

struct Fila {

    // Aponta para o primeiro nó da fila.
    struct No *inicio;

    // Aponta para o último nó da fila.
    struct No *final;

    // Quantidade de elementos atualmente na fila.
    int quantidade;
};


//=========================================================
// ENQUEUE
// Insere um elemento no final da fila.
//=========================================================

void enqueue(struct Fila *fila, int valor)
{
    // Cria um ponteiro para o novo nó.
    struct No *novo = NULL;

    // Reserva memória para o novo nó.
    novo = malloc(sizeof(struct No));

    // Insere o valor no novo nó.
    novo->dado = valor;

    // O novo nó será o último.
    // Portanto, não possui próximo.
    novo->proximo = NULL;


    // Verifica se a fila está vazia.
    if (fila->inicio == NULL) {

        // O novo nó será o primeiro.
        fila->inicio = novo;

        // O novo nó também será o último.
        fila->final = novo;

    } else {

        // O antigo último aponta para o novo.
        fila->final->proximo = novo;

        // O novo passa a ser o último.
        fila->final = novo;
    }

    // Aumenta a quantidade de elementos.
    fila->quantidade++;
}


//=========================================================
// DEQUEUE
// Remove e retorna o primeiro elemento.
//=========================================================

int dequeue(struct Fila *fila)
{
    // Verifica se a fila está vazia.
    if (fila->inicio == NULL) {

        printf("Fila vazia!\n");

        return -1;
    }

    // Guarda o endereço do primeiro nó.
    struct No *remover = fila->inicio;

    // Guarda o valor que será removido.
    int valor = remover->dado;

    // O segundo nó passa a ser o primeiro.
    fila->inicio = fila->inicio->proximo;

    // Se a fila ficou vazia,
    // o final também precisa ser NULL.
    if (fila->inicio == NULL) {

        fila->final = NULL;
    }

    // Libera a memória do antigo primeiro nó.
    free(remover);

    // Diminui a quantidade de elementos.
    fila->quantidade--;

    // Retorna o valor removido.
    return valor;
}


//=========================================================
// PEEK
// Consulta o primeiro elemento sem removê-lo.
//=========================================================

int peek(struct Fila *fila)
{
    // Verifica se a fila está vazia.
    if (fila->inicio == NULL) {

        printf("Fila vazia!\n");

        return -1;
    }

    // Retorna o valor armazenado no primeiro nó.
    return fila->inicio->dado;
}


//=========================================================
// SIZE
// Retorna a quantidade de elementos.
//=========================================================

int size(struct Fila *fila)
{
    // Retorna a quantidade armazenada na fila.
    return fila->quantidade;
}


//=========================================================
// ISEMPTY
// Verifica se a fila está vazia.
//=========================================================

int isEmpty(struct Fila *fila)
{
    // Retorna 1 se a fila estiver vazia.
    // Caso contrário, retorna 0.
    return fila->inicio == NULL;
}


//=========================================================
// PERCORRENDO A FILA
//=========================================================

void imprimir(struct Fila *fila)
{
    // Cria um ponteiro para percorrer a fila.
    struct No *atual = fila->inicio;

    printf("Fila: ");

    // Percorre a fila enquanto houver um nó.
    while (atual != NULL) {

        // Mostra o valor armazenado no nó atual.
        printf("%d ", atual->dado);

        // Avança para o próximo nó.
        atual = atual->proximo;
    }

    printf("\n");
}


//=========================================================
// MAIN
//=========================================================

int main()
{
    // Cria uma fila.
    struct Fila fila;

    // A fila começa vazia.
    fila.inicio = NULL;
    fila.final = NULL;
    fila.quantidade = 0;


    //=====================================================
    // INSERINDO ELEMENTOS
    //=====================================================

    // Insere 10 no final da fila.
    enqueue(&fila, 10);

    // Insere 20 no final da fila.
    enqueue(&fila, 20);

    // Insere 30 no final da fila.
    enqueue(&fila, 30);


    //=====================================================
    // MOSTRANDO A FILA
    //=====================================================

    imprimir(&fila);


    //=====================================================
    // PEEK - CONSULTANDO O PRIMEIRO ELEMENTO
    //=====================================================

    printf("Primeiro elemento: %d\n",
           peek(&fila));

    //=====================================================
    //SIZE - CONSULTANDO A QUANTIDADE
    //=====================================================

    printf("Quantidade de elementos: %d\n",
           size(&fila));


    //=====================================================
    //ISEMPTY - Verifica se a fila está vazia
    //=====================================================

    if (isEmpty(&fila)) {

        printf("A fila esta vazia.\n");

    } else {

        printf("A fila nao esta vazia.\n");
    }

    //=====================================================
    // dequeue - REMOVENDO O PRIMEIRO ELEMENTO
    //=====================================================

    printf("Removido: %d\n",
           dequeue(&fila));


    //=====================================================
    // MOSTRANDO A FILA NOVAMENTE
    //=====================================================

    imprimir(&fila);

    //=====================================================
    //PEEK - MOSTRANDO O NOVO PRIMEIRO
    //=====================================================

    printf("Primeiro elemento agora: %d\n",
           peek(&fila));


    //=====================================================
    //SIZE - MOSTRANDO A NOVA QUANTIDADE
    //=====================================================

    printf("Quantidade agora: %d\n",
           size(&fila));


    return 0;
}
