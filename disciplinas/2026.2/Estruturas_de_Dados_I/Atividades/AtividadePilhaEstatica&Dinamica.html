Atividade - Pilha Estática e Pilha Dinâmica

Orientação: utilize os mesmos conceitos e códigos trabalhados na aula. A pilha segue a regra LIFO (Last In, First Out): o último elemento inserido é o primeiro a ser removido. Nas questões de programação, mantenha a organização dos exemplos da aula e utilize funções simples.
📎 Ver código de referência - Pilha Estática 
#include <stdio.h>

#define MAX 5

//=========================================================
//ESTRUTURA DA PILHA
//=========================================================

struct Pilha {

    //Vetor que armazena os elementos.
    int dados[MAX];

    //Indica a posição do elemento que está no topo.
    int topo;
};


//=========================================================
//PUSH - INSERE UM ELEMENTO NO TOPO
//=========================================================

void push(struct Pilha *pilha, int valor)
{
    //Verifica se a pilha está cheia.
    if (pilha->topo == MAX - 1) {

        printf("Pilha cheia!\n");

        return;
    }

    //Avança o topo para a próxima posição.
    pilha->topo++;

    //Insere o valor na posição indicada pelo topo.
    pilha->dados[pilha->topo] = valor;
}


//=========================================================
//POP - REMOVE E RETORNA O ELEMENTO DO TOPO
//=========================================================

int pop(struct Pilha *pilha)
{
    //Verifica se a pilha está vazia.
    if (pilha->topo == -1) {

        printf("Pilha vazia!\n");

        return -1;
    }

    //Guarda o valor que será removido.
    int valor = pilha->dados[pilha->topo];

    //Move o topo para a posição anterior.
    pilha->topo--;

    //Retorna o valor removido.
    return valor;
}


//=========================================================
//PEEK - CONSULTA O ELEMENTO DO TOPO
//=========================================================

int peek(struct Pilha *pilha)
{
    //Verifica se a pilha está vazia.
    if (pilha->topo == -1) {

        printf("Pilha vazia!\n");

        return -1;
    }

    //Retorna o elemento que está no topo.
    return pilha->dados[pilha->topo];
}


//=========================================================
//SIZE - RETORNA A QUANTIDADE DE ELEMENTOS
//=========================================================

int size(struct Pilha *pilha)
{
    //Como o índice começa em 0,
    //a quantidade é topo + 1.
    return pilha->topo + 1;
}


//=========================================================
//ISEMPTY - VERIFICA SE A PILHA ESTÁ VAZIA
//=========================================================

int isEmpty(struct Pilha *pilha)
{
    //Retorna 1 se o topo for -1.
    return pilha->topo == -1;
}


//=========================================================
//ISFULL - VERIFICA SE A PILHA ESTÁ CHEIA
//=========================================================

int isFull(struct Pilha *pilha)
{
    //Retorna 1 se o topo estiver na última posição.
    return pilha->topo == MAX - 1;
}


//=========================================================
//MAIN
//=========================================================

int main()
{
    //Cria uma pilha.
    struct Pilha pilha;

    //Inicialmente, a pilha está vazia.
    pilha.topo = -1;

    //PUSH - Insere elementos.
    push(&pilha, 10);
    push(&pilha, 20);
    push(&pilha, 30);

    //=====================================================
    //PEEK - Consulta o elemento do topo.
    //=====================================================

    printf("Topo: %d\n", peek(&pilha));
    
    
    //Verificando se a pilha está vazia e/ou cheia.
    printf("A pilha está vazia?\n");

    // isEmpty() retorna 1 se a pilha estiver vazia.
        if (isEmpty(&pilha)) {
            printf("Pilha vazia! Valor lógico: %d\n", isEmpty(&pilha));
        } else {
            printf("Pilha não está vazia. Valor lógico: %d\n", isEmpty(&pilha));
        }
    
    // isFull() retorna 1 se a pilha estiver cheia.
    printf("A pilha está cheia?\n");
    
        if (isFull(&pilha)) {
            printf("Pilha cheia! Valor lógico: %d\n", isFull(&pilha));
        } else {
            printf("Pilha não está cheia. Valor lógico: %d\n", isFull(&pilha));
        }
        
    
    //=====================================================
    //SIZE - Retorna a quantidade de elementos.
    //=====================================================

    printf("Quantidade: %d\n", size(&pilha));

    //=====================================================
    //POP - Remove e retorna o elemento do topo.
    //=====================================================

    printf("Removido: %d\n", pop(&pilha));

    //PEEK - Mostra o novo topo.
    printf("Novo topo: %d\n", peek(&pilha));

    return 0;
}

#include <stdio.h>
#include <stdlib.h>

//=========================================================
//ESTRUTURA DO NÓ
//=========================================================

struct No {

    //Valor armazenado no nó.
    int dado;

    //Endereço do próximo nó.
    struct No *proximo;
};


//=========================================================
//ESTRUTURA DA PILHA
//=========================================================

struct Pilha {

    //Aponta para o nó que está no topo.
    struct No *topo;

    //Quantidade de elementos da pilha.
    int quantidade;
};


//=========================================================
//PUSH - INSERE UM ELEMENTO NO TOPO
//=========================================================

void push(struct Pilha *pilha, int valor)
{
    //Cria um ponteiro para o novo nó.
    struct No *novo = NULL;

    //Reserva memória para o novo nó.
    novo = malloc(sizeof(struct No));

    //Insere o valor no novo nó.
    novo->dado = valor;

    //O novo nó aponta para o antigo topo.
    novo->proximo = pilha->topo;

    //O novo nó passa a ser o topo.
    pilha->topo = novo;

    //Aumenta a quantidade de elementos.
    pilha->quantidade++;
}


//=========================================================
//POP - REMOVE E RETORNA O ELEMENTO DO TOPO
//=========================================================

int pop(struct Pilha *pilha)
{
    //Verifica se a pilha está vazia.
    if (pilha->topo == NULL) {

        printf("Pilha vazia!\n");

        return -1;
    }

    //Guarda o endereço do nó que será removido.
    struct No *remover = pilha->topo;

    //Guarda o valor que será removido.
    int valor = remover->dado;

    //O próximo nó passa a ser o topo.
    pilha->topo = pilha->topo->proximo;

    //Libera a memória do nó removido.
    free(remover);

    //Diminui a quantidade de elementos.
    pilha->quantidade--;

    //Retorna o valor removido.
    return valor;
}


//=========================================================
//PEEK - CONSULTA O ELEMENTO DO TOPO
//=========================================================

int peek(struct Pilha *pilha)
{
    //Verifica se a pilha está vazia.
    if (pilha->topo == NULL) {

        printf("Pilha vazia!\n");

        return -1;
    }

    //Retorna o valor do topo.
    return pilha->topo->dado;
}


//=========================================================
//SIZE - RETORNA A QUANTIDADE DE ELEMENTOS
//=========================================================

int size(struct Pilha *pilha)
{
    //Retorna a quantidade armazenada.
    return pilha->quantidade;
}


//=========================================================
//ISEMPTY - VERIFICA SE A PILHA ESTÁ VAZIA
//=========================================================

int isEmpty(struct Pilha *pilha)
{
    //Retorna 1 se não houver nenhum nó.
    return pilha->topo == NULL;
}


//=========================================================
//IMPRIMIR - MOSTRA OS ELEMENTOS DA PILHA
//=========================================================

void imprimir(struct Pilha *pilha)
{
    //Começa pelo topo.
    struct No *atual = pilha->topo;

    //Percorre a pilha até encontrar NULL.
    while (atual != NULL) {

        printf("[%d] ", atual->dado);

        atual = atual->proximo;
    }

    printf("\n");
}


//=========================================================
//MAIN
//=========================================================

int main()
{
    //Cria uma pilha.
    struct Pilha pilha;

    //Inicialmente, não existe nenhum nó.
    pilha.topo = NULL;

    //Inicialmente, a quantidade é zero.
    pilha.quantidade = 0;

    //PUSH - Insere elementos.
    push(&pilha, 10);
    push(&pilha, 20);
    push(&pilha, 30);

     //isEmpty - Verifica se a pilha está vazia.
    printf("A pilha está vazia?\n");
    
    if (isEmpty(&pilha)){
        printf("A pilha está vazia!Valor lógico: %d ('True')\n", isEmpty(&pilha));
    } else {
        printf("A pilha não está vazia!\nEla contém: %d elementos.\nValor lógico retornado da função ('isEmpty()'): %d ('False')\n", size(&pilha), isEmpty(&pilha));
    }
    
    //PEEK - Consulta o elemento do topo.
    printf("Topo: %d\n", peek(&pilha));
    
    //IMPRIMIR - Mostra os elementos da pilha.
    printf("Elementos da pilha: ");
    imprimir(&pilha);

    //POP - Remove e retorna o elemento do topo que foi removido.
    printf("Removido: %d\n", pop(&pilha));

    //PEEK - Consulta o elemento do topo. Mostra o novo topo.
    printf("Novo topo: %d\n", peek(&pilha));
    
    //SIZE - Retorna a quantidade de elementos atuais.
    printf("Quantidade atual: %d\n", size(&pilha));
    
    //IMPRIMIR - Mostra os elementos da pilha após remoção.
    printf("Elementos da pilha atual: ");
    imprimir(&pilha);

    return 0;
}
