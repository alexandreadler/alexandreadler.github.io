Atividade - Lista Circular

📎 Ver Código de Referência (Lista Circular) 
#include <stdio.h>
#include <stdlib.h>

int main()
{
    //=========================================================
    //1. DEFININDO A ESTRUTURA DO NÓ
    //=========================================================

    /*
     * Cada nó possui:
     *
     * dado    -> valor armazenado
     * proximo -> endereço do próximo nó
     *
     * Na lista circular, o último nó aponta
     * novamente para o primeiro.
     */
    struct No {
        int dado;
        struct No *proximo;
    };


    //=========================================================
    //2. CRIANDO OS PONTEIROS PARA OS NÓS
    //=========================================================

    struct No *primeiro = NULL;
    struct No *segundo = NULL;
    struct No *terceiro = NULL;
    struct No *quarto = NULL;


    //=========================================================
    //3. ALOCANDO MEMÓRIA
    //=========================================================

    primeiro = malloc(sizeof(struct No));
    segundo = malloc(sizeof(struct No));
    terceiro = malloc(sizeof(struct No));
    quarto = malloc(sizeof(struct No));


    //=========================================================
    //4. INSERINDO OS VALORES
    //=========================================================

    primeiro->dado = 10;
    segundo->dado = 20;
    terceiro->dado = 30;
    quarto->dado = 40;


    //=========================================================
    //5. FAZENDO AS LIGAÇÕES
    //=========================================================

    primeiro->proximo = segundo;
    segundo->proximo = terceiro;
    terceiro->proximo = quarto;

    /*
     * Aqui está a principal característica
     * da lista circular:
     *
     * o último aponta novamente para o primeiro.
     */
    quarto->proximo = primeiro;


    //=========================================================
    //6. PERCORRENDO A LISTA
    //=========================================================

    struct No *atual = primeiro;

    /*
     * Não podemos usar:
     *
     * while (atual != NULL)
     *
     * porque nunca chegaremos a NULL.
     *
     * Paramos quando voltarmos ao primeiro nó.
     */

    printf("Lista inicial: ");

    do {

        printf("%d ", atual->dado);

        atual = atual->proximo;

    } while (atual != primeiro);


    //=========================================================
    //7. BUSCA
    //=========================================================

    int valor = 30;

    // Fazemos o ponteiro atual voltar para o primeiro nó.
    atual = primeiro;

    int encontrado = 0;

    do {

        if (atual->dado == valor) {

            printf("\nElemento %d encontrado!\n", valor);

            encontrado = 1;

            break;
        }

        atual = atual->proximo;

    } while (atual != primeiro);


    if (encontrado == 0) {

        printf("\nElemento %d nao encontrado.\n", valor);
    }


    //=========================================================
    //8. INSERÇÃO NO INÍCIO
    //=========================================================

    /*
     * Antes:
     *
     * [10] -> [20] -> [30] -> [40]
     *  ↑                         │
     *  └─────────────────────────┘
     *
     * Queremos inserir 5.
     */

    struct No *novo = NULL;

    novo = malloc(sizeof(struct No));

    novo->dado = 5;

    /*
     * O novo nó aponta para o antigo primeiro.
     */
    novo->proximo = primeiro;

    /*
     * O antigo último precisa apontar
     * para o novo primeiro.
     */
    quarto->proximo = novo;

    /*
     * Agora o novo nó passa a ser o primeiro.
     */
    primeiro = novo;

    printf("\nDepois de inserir 5 no inicio: ");

    atual = primeiro;

    do {

        printf("%d ", atual->dado);

        atual = atual->proximo;

    } while (atual != primeiro);

    printf("\n");


    //=========================================================
    //9. INSERÇÃO NO FINAL
    //=========================================================

    /*
     * Vamos inserir 50 depois do 40.
     */

    struct No *ultimo = NULL;

    ultimo = malloc(sizeof(struct No));

    ultimo->dado = 50;

    /*
     * O novo nó aponta para o primeiro,
     * pois ele será o último da lista circular.
     */
    ultimo->proximo = primeiro;

    /*
     * O antigo último (40) passa a apontar
     * para o novo último (50).
     */
    quarto->proximo = ultimo;

    printf("Depois de inserir 50 no final: ");

    // Fazemos o ponteiro atual voltar para o primeiro nó.
    atual = primeiro;

    do {

        printf("%d ", atual->dado);

        atual = atual->proximo;

    } while (atual != primeiro);

    printf("\n");


    //=========================================================
    //10. REMOÇÃO DO NÓ 20
    //=========================================================

    /*
     * Queremos remover o nó 20.
     */

    struct No *remover = segundo;

    /*
     * Como esta é uma lista simplesmente encadeada,
     * o nó 20 não possui um campo 'anterior'.
     *
     * Precisamos encontrar quem aponta para ele.
     */
    struct No *anterior = primeiro;

    /*
     * Percorremos a lista até encontrar
     * o nó cujo próximo é o nó que queremos remover.
     */
    while (anterior->proximo != remover) {

        anterior = anterior->proximo;
    }

    /*
     * Agora:
     *
     * anterior -> [10]
     * remover  -> [20]
     *
     * Fazemos o 10 apontar para o próximo do 20.
     */
    anterior->proximo = remover->proximo;

    /*
     * Agora o 20 não faz mais parte da lista.
     *
     * Liberamos sua memória.
     */
    free(remover);

    printf("Depois de remover 20: ");

    // Fazemos o ponteiro atual voltar para o primeiro nó.
    atual = primeiro;

    do {

        printf("%d ", atual->dado);

        atual = atual->proximo;

    } while (atual != primeiro);

    printf("\n");


    //=========================================================
    //11. FINALIZANDO
    //=========================================================

    return 0;
}

Objetivo: Praticar a criação de uma lista circular simples, suas ligações entre os nós, o percurso, a busca, a inserção no início e no final e a remoção de um nó.

Parte 1 - Entendendo a lista circular

Observe a ligação responsável por transformar a lista em circular:

primeiro->proximo = segundo;
segundo->proximo = terceiro;
terceiro->proximo = quarto;

quarto->proximo = primeiro;

Parte 2 - Por que não usamos NULL?

Na lista circular, o percurso é feito com:

do {
    printf("%d ", atual->dado);
    atual = atual->proximo;
} while (atual != primeiro);
