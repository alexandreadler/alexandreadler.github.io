Atividade - Lista Duplamente Encadeada

Orientação: nesta atividade, continue utilizando o mesmo código de exemplo da aula (exemploAula8_5.3-ListaDuplamenteEncadeada.c): um único arquivo C, sem criar funções e com a inserção dos nós feita manualmente. O objetivo é entender os ponteiros anterior e proximo.
📎 Ver código de referência da aula 
#include <stdio.h>
#include <stdlib.h>

int main()
{
    // =========================================================
    // 1. DEFININDO A ESTRUTURA DO NÓ
    // =========================================================

    /*
     * Cada nó possui três informações:
     *
     * dado      -> valor armazenado no nó
     * anterior  -> endereço do nó anterior
     * proximo   -> endereço do próximo nó
     */
    struct No {
        int dado;
        struct No *anterior;
        struct No *proximo;
    };


    // =========================================================
    // 2. CRIANDO OS PONTEIROS PARA OS NÓS
    // =========================================================

    /*
     * Criamos quatro ponteiros para representar os nós.
     * Inicialmente, eles não apontam para nenhum nó.
     */
    struct No *primeiro = NULL;
    struct No *segundo = NULL;
    struct No *terceiro = NULL;
    struct No *quarto = NULL;


    // =========================================================
    // 3. ALOCANDO MEMÓRIA PARA OS NÓS
    // =========================================================

    /*
     * malloc() reserva na memória o espaço necessário
     * para armazenar um struct No.
     */
    primeiro = malloc(sizeof(struct No));
    segundo = malloc(sizeof(struct No));
    terceiro = malloc(sizeof(struct No));
    quarto = malloc(sizeof(struct No));


    // =========================================================
    // 4. INSERINDO OS VALORES NOS NÓS
    // =========================================================

    /*
     * O operador -> permite acessar um campo da estrutura
     * através de um ponteiro.
     */
    primeiro->dado = 10;
    segundo->dado = 20;
    terceiro->dado = 30;
    quarto->dado = 40;


    // =========================================================
    // 5. FAZENDO AS LIGAÇÕES DA LISTA
    // =========================================================

    /*
     * Queremos construir:
     *
     * NULL <- [10] <-> [20] <-> [30] <-> [40] -> NULL
     */

    // O primeiro não possui nó anterior.
    primeiro->anterior = NULL;

    // O primeiro aponta para o segundo.
    primeiro->proximo = segundo;

    // O segundo aponta de volta para o primeiro.
    segundo->anterior = primeiro;

    // O segundo aponta para o terceiro.
    segundo->proximo = terceiro;

    // O terceiro aponta de volta para o segundo.
    terceiro->anterior = segundo;

    // O terceiro aponta para o quarto.
    terceiro->proximo = quarto;

    // O quarto aponta de volta para o terceiro.
    quarto->anterior = terceiro;

    // O quarto é o último, portanto não possui próximo.
    quarto->proximo = NULL;


    // =========================================================
    // 6. PERCORRENDO A LISTA PARA FRENTE
    // =========================================================

    /*
     * 'atual' é um ponteiro auxiliar.
     * Ele começa apontando para o primeiro nó.
     */
    struct No *atual = primeiro;

    while (atual != NULL) {

        // Mostra o valor armazenado no nó atual.
        printf("%d ", atual->dado);

        // Avança para o próximo nó.
        atual = atual->proximo;
    }


    // =========================================================
    // 7. PERCORRENDO A LISTA PARA TRÁS
    // =========================================================

    // Começamos pelo último nó.
    atual = quarto;

    while (atual != NULL) {

        // Mostra o valor do nó atual.
        printf("%d ", atual->dado);

        // Volta para o nó anterior.
        atual = atual->anterior;
    }


    // =========================================================
    // 8. BUSCA
    // =========================================================

    // Valor que queremos procurar.
    int valor = 30;

    // A busca começa no primeiro nó.
    atual = primeiro;

    /*
     * 0 = não encontrado
     * 1 = encontrado
     */
    int encontrado = 0;

    while (atual != NULL) {

        // Verifica se o valor do nó atual é o procurado.
        if (atual->dado == valor) {

            printf("Elemento %d encontrado no nó: %p!\n",
                   valor, atual);

            // Marca que encontramos o elemento.
            encontrado = 1;
        }

        /*
         * Continua a busca.
         * Não usamos break porque podemos querer encontrar
         * todas as ocorrências do valor.
         */
        atual = atual->proximo;
    }

    // Se percorremos toda a lista e não encontramos o valor...
    if (encontrado == 0)
        printf("Elemento %d não encontrado.\n", valor);


    // =========================================================
    // 9. INSERÇÃO NO INÍCIO
    // =========================================================

    /*
     * Lista antes:
     *
     * [10] <-> [20] <-> [30] <-> [40]
     *
     * Queremos inserir 5:
     *
     * [5] <-> [10] <-> [20] <-> [30] <-> [40]
     */

    // Ponteiro que será usado para o novo nó.
    struct No *quinto = NULL;

    // Reserva memória para o novo nó.
    quinto = malloc(sizeof(struct No));

    // Insere 5 no novo nó.
    quinto->dado = 5;

    // Como será o primeiro, não possui anterior.
    quinto->anterior = NULL;

    // O novo nó aponta para o antigo primeiro.
    quinto->proximo = primeiro;

    // O antigo primeiro passa a apontar de volta para o novo.
    primeiro->anterior = quinto;

    /*
     * A ligação já foi criada, mas 'primeiro' ainda apontava
     * para o nó 10.
     *
     * Agora atualizamos o ponteiro que representa o início.
     */
    primeiro = quinto;


    // =========================================================
    // 10. INSERÇÃO NO FINAL
    // =========================================================

    /*
     * Queremos inserir 50 depois do 40:
     *
     * [5] <-> [10] <-> [20] <-> [30] <-> [40] <-> [50]
     */

    // Ponteiro que será usado para o novo último nó.
    struct No *sexto = NULL;

    // Reserva memória para o novo nó.
    sexto = malloc(sizeof(struct No));

    // Insere 50 no novo nó.
    sexto->dado = 50;

    // Como será o último, não possui próximo.
    sexto->proximo = NULL;

    // O nó anterior ao novo último é o nó 40.
    sexto->anterior = quarto;

    // O antigo último (40) passa a apontar para o novo (50).
    quarto->proximo = sexto;


    // =========================================================
    // 11. REMOÇÃO DO NÓ 20
    // =========================================================

    /*
     * Antes:
     *
     * [5] <-> [10] <-> [20] <-> [30] <-> [40] <-> [50]
     *                  ↑
     *               remover
     */

    /*
     * 'segundo' aponta para o nó 20.
     *
     * NÃO estamos fazendo uma cópia do nó.
     * Estamos apenas criando outro ponteiro que aponta
     * para o mesmo nó.
     *
     * segundo ----\
     *              > [20]
     * remover ----/
     */
    struct No *remover = segundo;


    /*
     * O nó anterior ao 20 é o 10.
     * O nó seguinte ao 20 é o 30.
     *
     * Esta linha faz:
     *
     *     10->proximo = 30
     *
     * O 10 passa a "pular" o 20.
     */
    remover->anterior->proximo = remover->proximo;


    /*
     * Agora corrigimos a ligação no sentido contrário.
     *
     * Esta linha faz:
     *
     *     30->anterior = 10
     *
     * Assim, 10 e 30 passam a estar ligados diretamente
     * nos dois sentidos.
     */
    remover->proximo->anterior = remover->anterior;


    /*
     * As ligações já foram corrigidas.
     * Agora podemos liberar a memória do nó 20.
     */
    free(remover);


    /*
     * Lista final:
     *
     * [5] <-> [10] <-> [30] <-> [40] <-> [50]
     */


    // =========================================================
    // 12. FINALIZANDO O PROGRAMA
    // =========================================================

    return 0;
}
