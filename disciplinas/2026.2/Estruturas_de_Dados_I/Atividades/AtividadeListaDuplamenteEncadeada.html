/*
 * LISTA DUPLAMENTE ENCADEADA
 * Exemplo simples - sem uso de funções
 *
 * A lista será construída manualmente:
 *
 * NULL <- [10] <-> [20] <-> [30] <-> [40] -> NULL
 */

#include <stdio.h>
#include <stdlib.h> //Necessário para usar malloc() e free()

int main()
{
    //=========================================================
    //1. DEFININDO A ESTRUTURA DO NÓ
    //=========================================================

    /*
     * Cada nó possui três partes:
     *
     * anterior -> endereço do nó anterior
     * dado      -> valor armazenado
     * proximo   -> endereço do próximo nó
     */
    struct No {
        int dado;
        struct No *anterior;
        struct No *proximo;
    };


    //=========================================================
    //2. CRIANDO OS PONTEIROS
    //=========================================================

    //Ponteiro que representará o primeiro nó.
    struct No *primeiro = NULL;

    //Ponteiro que representará o segundo nó.
    struct No *segundo = NULL;

    //Ponteiro que representará o terceiro nó.
    struct No *terceiro = NULL;

    //Ponteiro que representará o quarto nó.
    struct No *quarto = NULL;

    /*
     * Neste momento:
     *
     * primeiro -> NULL
     * segundo  -> NULL
     * terceiro -> NULL
     * quarto   -> NULL
     *
     * Ainda não criamos nenhum nó.
     */


    //=========================================================
    //3. ALOCANDO MEMÓRIA PARA OS NÓS
    //=========================================================

    //Reserva espaço na memória para o primeiro nó.
    primeiro = malloc(sizeof(struct No));

    //Reserva espaço na memória para o segundo nó.
    segundo = malloc(sizeof(struct No));

    //Reserva espaço na memória para o terceiro nó.
    terceiro = malloc(sizeof(struct No));

    //Reserva espaço na memória para o quarto nó.
    quarto = malloc(sizeof(struct No));


    //=========================================================
    //4. INSERINDO OS DADOS NOS NÓS
    //=========================================================

    //O primeiro nó armazenará o valor 10.
    primeiro->dado = 10;

    //O segundo nó armazenará o valor 20.
    segundo->dado = 20;

    //O terceiro nó armazenará o valor 30.
    terceiro->dado = 30;

    //O quarto nó armazenará o valor 40.
    quarto->dado = 40;


    //=========================================================
    //5. FAZENDO AS LIGAÇÕES
    //=========================================================

    /*
     * Vamos construir:
     *
     * NULL <- [10] <-> [20] <-> [30] <-> [40] -> NULL
     */


    //---------------------------------------------------------
    //PRIMEIRO NÓ
    //---------------------------------------------------------

    //O primeiro nó não possui anterior.
    primeiro->anterior = NULL;

    //O primeiro nó aponta para o segundo.
    primeiro->proximo = segundo;


    //---------------------------------------------------------
    //SEGUNDO NÓ
    //---------------------------------------------------------

    //O segundo aponta de volta para o primeiro.
    segundo->anterior = primeiro;

    //O segundo aponta para o terceiro.
    segundo->proximo = terceiro;


    //---------------------------------------------------------
    //TERCEIRO NÓ
    //---------------------------------------------------------

    //O terceiro aponta de volta para o segundo.
    terceiro->anterior = segundo;

    //O terceiro aponta para o quarto.
    terceiro->proximo = quarto;


    //---------------------------------------------------------
    //QUARTO NÓ
    //---------------------------------------------------------

    //O quarto aponta de volta para o terceiro.
    quarto->anterior = terceiro;

    //O quarto é o último nó, portanto seu próximo é NULL.
    quarto->proximo = NULL;


    //=========================================================
    //6. MOSTRANDO A LISTA 'Manualmente'
    //=========================================================

    printf("Lista duplamente encadeada:\n");

    printf("NULL <- %d <-> %d <-> %d <-> %d -> NULL\n",
           primeiro->dado,
           segundo->dado,
           terceiro->dado,
           quarto->dado);


    //=========================================================
    //7. PERCORRENDO A LISTA PARA FRENTE
    //=========================================================

    /*
     * Criamos um ponteiro auxiliar.
     *
     * Ele começará no primeiro nó e irá avançando
     * utilizando o campo proximo.
     */
    struct No *atual = primeiro;

    printf("\nPercorrendo para frente:\n");

    //Continua enquanto atual não for NULL.
    while (atual != NULL) {

        //Mostra o valor do nó atual.
        printf("%d ", atual->dado);

        //Avança para o próximo nó.
        atual = atual->proximo;
    }

    printf("\n");


    //=========================================================
    //8. PERCORRENDO A LISTA PARA TRÁS
    //=========================================================

    /*
     * Agora começamos no último nó.
     *
     * O quarto é o último nó da nossa lista.
     */
    atual = quarto;

    printf("\nPercorrendo para tras:\n");

    //Continua enquanto atual não for NULL.
    while (quarto != NULL) {

        //Mostra o valor do nó atual.
        printf("%d ", atual->dado);

        //Volta para o nó anterior.
        atual = atual->anterior;
    }

    printf("\n");


    //=========================================================
    //9. ACESSANDO UM NÓ A PARTIR DO PRIMEIRO.
    //=========================================================

    printf("\nAcessando os nos a partir do primeiro:\n");

    //Acessa o valor do primeiro nó.
    printf("Primeiro: %d\n", primeiro->dado);

    //Vai do primeiro para o segundo.
    printf("Segundo: %d\n", primeiro->proximo->dado);

    //Vai do primeiro para o segundo e depois para o terceiro.
    printf("Terceiro: %d\n",
           primeiro->proximo->proximo->dado);

    //Vai do primeiro até o quarto.
    printf("Quarto: %d\n",
           primeiro->proximo->proximo->proximo->dado);


    //=========================================================
    //10. ACESSANDO PARA TRÁS
    //=========================================================

    printf("\nAcessando os nos a partir do ultimo:\n");

    //Acessa o valor do quarto nó.
    printf("Quarto: %d\n", quarto->dado);

    //Vai do quarto para o terceiro.
    printf("Terceiro: %d\n", quarto->anterior->dado);

    //Vai do quarto para o terceiro e depois para o segundo.
    printf("Segundo: %d\n",
           quarto->anterior->anterior->dado);

    //Vai do quarto até o primeiro.
    printf("Primeiro: %d\n",
           quarto->anterior->anterior->anterior->dado);


    //=========================================================
    //11. BUSCA MANUAL
    //=========================================================

    /*
     * Vamos procurar o valor 30.
     *
     * O ponteiro atual começa no primeiro nó.
     */
    int valor = 30;

    atual = primeiro;

    int encontrado = 0;

    printf("\nBuscando o valor %d...\n", valor);

    //Percorre a lista enquanto houver nós.
    while (atual != NULL) {

        //Verifica se o valor atual é o procurado.
        if (atual->dado == valor) {

            printf("Elemento %d encontrado no nó: %p!\n", valor, atual);

            encontrado = 1;

        }

        //Avança para o próximo nó.
        atual = atual->proximo;
    }
        if (encontrado == 0) {
            
            printf("Elemento %d não encontrado.\n", valor);
    }


    //=========================================================
    //12. INSERÇÃO NO INÍCIO - EXEMPLO
    //=========================================================

    /*
     * Lista original:
     *
     * NULL <- [10] <-> [20] <-> [30] <-> [40] -> NULL
     *
     * Queremos inserir 5:
     *
     * NULL <- [5] <-> [10] <-> [20] <-> [30] <-> [40] -> NULL
     */

    //Cria um ponteiro para o novo nó.
    struct No *novo = NULL;

    //Reserva memória para o novo nó.
    novo = malloc(sizeof(struct No));

    //Insere o valor 5 no novo nó.
    novo->dado = 5;

    //O novo nó será o primeiro.
    //Portanto, não possui anterior.
    novo->anterior = NULL;

    //O novo nó aponta para o antigo primeiro. O próximo do nó '5' será o nó que atualmente está em primeiro
    //Como primeiro aponta para 10. Agora 5 'conhece' 10.
    novo->proximo = primeiro;

    //O antigo primeiro passa a apontar de volta para o novo. O nó 10 agora sabe que antes dele está o nó 5.
    //Ou seja, no nó que atualmente é apontado por 'primeiro', insira 'novo' no campo 'anterior'.
    primeiro->anterior = novo;

    //O novo nó passa a ser o primeiro. Variável que representa o primeiro nó da lista agora aponta para o nó 5
    //Essa instrução não cria uma nova ligação entre os nós. Ela muda para onde a variável primeiro aponta.
    primeiro = novo;


    //Mostra a lista depois da inserção.
    atual = primeiro;

    printf("\nDepois de inserir 5 no inicio:\n");

    while (atual != NULL) {

        printf("%d ", atual->dado);

        atual = atual->proximo;
    }

    printf("\n");


    //=========================================================
    //13. INSERÇÃO NO FINAL - EXEMPLO
    //=========================================================

    /*
     * Agora queremos inserir 50 no final.
     *
     * Lista:
     *
     * [5] <-> [10] <-> [20] <-> [30] <-> [40]
     *
     * Resultado:
     *
     * [5] <-> [10] <-> [20] <-> [30] <-> [40] <-> [50]
     */

    //Cria um novo nó.
    struct No *ultimo = NULL;

    //Reserva memória para o novo nó.
    ultimo = malloc(sizeof(struct No));

    //Insere o valor 50 no novo nó.
    ultimo->dado = 50;

    //O novo nó será o último.
    //Portanto, não possui próximo.
    ultimo->proximo = NULL;

    /*
     * O antigo último é o quarto nó.
     *
     * O novo nó precisa apontar para ele.
     */
    ultimo->anterior = quarto;

    /*
     * O antigo último precisa apontar para o novo.
     */
    quarto->proximo = ultimo;


    //Mostra a lista depois da inserção.
    atual = primeiro;

    printf("\nDepois de inserir 50 no final:\n");

    while (atual != NULL) {

        printf("%d ", atual->dado);

        atual = atual->proximo;
    }

    printf("\n");


    //=========================================================
    //14. REMOÇÃO DO NÓ 20 - EXEMPLO
    //=========================================================

    /*
     * Antes:
     *
     * [5] <-> [10] <-> [20] <-> [30] <-> [40] <-> [50]
     *
     * Queremos remover o nó 20.
     *
     * O nó 10 precisa apontar para o 30.
     *
     * O nó 30 precisa apontar de volta para o 10.
     */

    //O nó que será removido é o segundo nó.
    //'remover' aponta para o 20.
    struct No *remover = segundo;

    //O nó anterior ao removido passa a apontar para o próximo.
    //Ou seja, o anterior do nó que vou remover deve apontar para o próximo.
    remover->anterior->proximo = remover->proximo;

    //O nó seguinte ao removido passa a apontar para o anterior.
    //Ou seja, o próximo do nó que vou remover deve apontar para o anterior.
    remover->proximo->anterior = remover->anterior;

    //Libera a memória do nó removido.
    free(remover);


    //Mostra a lista depois da remoção.
    atual = primeiro;

    printf("\nDepois de remover 20:\n");

    while (atual != NULL) {

        printf("%d ", atual->dado);

        atual = atual->proximo;
    }

    printf("\n");


    //=========================================================
    //15. LIBERANDO A MEMÓRIA
    //=========================================================

    /*
     * Ainda temos os nós:
     *
     * 5, 10, 30, 40 e 50
     *
     * Precisamos liberar a memória de cada um.
     */

    free(primeiro);  //Libera o nó 5

    free(terceiro);  //Libera o nó 30

    free(quarto);    //Libera o nó 40

    free(ultimo);    //Libera o nó 50


    //Finaliza o programa.
    return 0;
}
