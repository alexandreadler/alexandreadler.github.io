#include <stdio.h>
#include <stdlib.h> // Necessário para usar malloc

// Definindo a estrutura do nó
struct No {
    int dado;
    struct No* proximo;
};

int main() {
    // 1. Criando os ponteiros para os nós
    struct No* primeiro = NULL;
    struct No* segundo = NULL;
    struct No* terceiro = NULL;

    // 2. Alocando memória para cada nó no computador
    primeiro = (struct No*)malloc(sizeof(struct No));
    segundo = (struct No*)malloc(sizeof(struct No));
    terceiro = (struct No*)malloc(sizeof(struct No));

    // 3. Inserindo valores dentro dos nós
    primeiro->dado = 10; // O primeiro nó guarda o valor 10
    segundo->dado = 20;  // O segundo nó guarda o valor 20
    terceiro->dado = 30;  // O terceiro nó guarda o valor 30

    // 4. FAZENDO A LIGAÇÃO
    // O 'proximo' do primeiro nó aponta para o endereço do segundo nó
    primeiro->proximo = segundo;

    // O segundo nó é o último, então o 'proximo' dele aponta para o vazio (NULL)
    //segundo->proximo = NULL;
    
    //Nó criado para demonstrar o slide "Percorrendo um lista"
    //'Descomentar' durante a explicação do slide
    segundo->proximo = terceiro;
    
    terceiro->proximo = NULL;

    // 5. Imprimindo os valores navegando pela estrutura
    printf("Valor do primeiro no: %d\n", primeiro->dado);
    
    // Conseguimos acessar o valor do segundo nó através do primeiro!
    printf("Valor do segundo no (via primeiro): %d\n", primeiro->proximo->dado);

    // Conseguimos acessar o valor do segundo nó através do primeiro!
    printf("Valor do terceiro no (via segundo): %d\n", segundo->proximo->dado);
    
    // ====================================================================
    // 6. EXIBINDO A LISTA COMPLETA (Percorrendo a estrutura do início ao fim)
    // ====================================================================

    // Ponteiro auxiliar que servirá como o nosso "navegador" ou "cursor".
    // Usamos ele para não perder a referência de onde a lista começa (o 'primeiro').
    struct No *atual;

    // Posiciona o nosso cursor no início da lista (no primeiro nó)
    atual = primeiro;

    // O laço continuará rodando enquanto o cursor estiver apontando para um nó válido.
    // Ele só vai parar quando chegar no 'proximo' do último nó, que vale NULL.
    while (atual != NULL) {

        // Exibe o valor armazenado dentro do nó onde o cursor está parado agora
        printf("%d\n", atual->dado);

        // Atualiza o 'cursor'. Ele deixa de apontar para o nó atual
        // e passa a apontar para o endereço do próximo nó da sequência.
        atual = atual->proximo;
    }


    // 7. Percorre enquanto houver nós com o objetivo de Buscar um valor na lista
    int valor;
    
    printf("Digite o valor para buscar: ");
    scanf("%d", &valor);
    
    // Reinicia e começa pelo primeiro nó
    atual = primeiro;
    
    while (atual != NULL) {
    
        // Verifica se encontrou o valor
        if (atual->dado == valor) {
    
            printf("Elemento %d encontrado no nó: %p!\n", valor, atual);
    
            break;
            
        } else{
            
            printf("Elemento %d Não encontrado no 'nó': %p!\n", valor, atual);  
            
        }
    
        // Avança para o próximo nó
        atual = atual->proximo;
    }
    // 8. Liberando a memória alocada
    free(primeiro);
    free(segundo);
    free(terceiro);

    return 0;
}
